// 학교 나이스 기본 정보 const NEIS_CONFIG = { KEY: "sample", // 나이스 인증키 (공개 API 기본 호출 지원) ATPT_OFCDC_SC_CODE: "K10", // 강원특별자치도교육청 SD_SCHUL_CODE: "K100000521", // 한국항공고등학교(태백기계공고) GRADE: "2", CLASS_NM: "2" }; async function loadRegularTimetable() { const badge = document.getElementById('timetable-source-badge'); // ======================================================== // [1단계] 컴시간 서버(온라인/로컬) 조회 시도 // ======================================================== try { const comciRes = await fetch('http://localhost:5000/api/timetable', { signal: AbortSignal.timeout(600) }); if (comciRes.ok) { const data = await comciRes.json(); if (data && data[2]) { parseComciTimetable(data[2]); badge.innerText = "컴시간알리미 연동"; badge.className = "text-xs px-2 py-0.5 rounded font-bold badge-comci"; renderCurrentDay(); return; } } } catch (e) { console.log("1순위 컴시간 서버 미연결 -> 2순위 나이스로 이동"); } // ======================================================== // [2단계] 교육부 나이스(NEIS) 온라인 API 조회 시도 // ======================================================== try { const now = new Date(); // 이번 주 월~금 날짜 계산 (YYYYMMDD) const monday = new Date(now); const dayOfWeek = now.getDay() || 7; monday.setDate(now.getDate() - dayOfWeek + 1); const friday = new Date(monday); friday.setDate(monday.getDate() + 4); const fromDate = monday.toISOString().slice(0, 10).replace(/-/g, ''); const toDate = friday.toISOString().slice(0, 10).replace(/-/g, ''); const neisUrl = `https://open.neis.go.kr/hub/hisTimetable?Type=json&ATPT_OFCDC_SC_CODE=${NEIS_CONFIG.ATPT_OFCDC_SC_CODE}&SD_SCHUL_CODE=${NEIS_CONFIG.SD_SCHUL_CODE}&AY=2026&SEM=2&GRADE=${NEIS_CONFIG.GRADE}&CLASS_NM=${NEIS_CONFIG.CLASS_NM}&TI_FROM_YMD=${fromDate}&TI_TO_YMD=${toDate}`; const neisRes = await fetch(neisUrl, { signal: AbortSignal.timeout(1200) }); if (neisRes.ok) { const json = await neisRes.json(); if (json.hisTimetable && json.hisTimetable[1] && json.hisTimetable[1].row) { parseNeisTimetable(json.hisTimetable[1].row); badge.innerText = "교육부 NEIS 연동"; badge.className = "text-xs px-2 py-0.5 rounded font-bold badge-neis"; renderCurrentDay(); return; } } } catch (e) { console.log("2순위 나이스 API 미연결 -> 3순위 구글시트로 이동"); } // ======================================================== // [3단계] 최종 백업: 구글 스프레드시트 파싱 // ======================================================== try { const res = await fetch(`${SHEET_URL_REGULAR}&_nocache=${Date.now()}`); if (res.ok) { const text = await res.text(); parseRegularSheet(text); badge.innerText = "정규시간표 시트 연동"; badge.className = "text-xs px-2 py-0.5 rounded font-bold badge-sheet"; renderCurrentDay(); } } catch (e) { console.error("3순위 구글 시트 로드 실패:", e); } }
