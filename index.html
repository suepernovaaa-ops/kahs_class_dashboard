""" 한국항공고등학교 2학년 2반 스마트 대시보드 백엔드 서버 - 컴시간알리미 다중 키워드 자동 순회 검색 및 파싱 - 시스템 트레이 상주 및 윈도우 시작 프로그램 자동 등록 - 로컬/교실 네트워크 다중 IP 지원 """ import datetime import json import os import re import sys import threading import urllib.parse import winreg from flask import Flask, jsonify, request from flask_cors import CORS import requests from PIL import Image, ImageDraw import pystray # ------------------------------------------------------------- # 1. 윈도우 시작 프로그램 자동 등록 # ------------------------------------------------------------- def register_startup(): try: exe_path = os.path.abspath(sys.argv[0]) key = winreg.OpenKey( winreg.HKEY_CURRENT_USER, r'Software\Microsoft\Windows\CurrentVersion\Run', 0, winreg.KEY_SET_VALUE ) winreg.SetValueEx(key, 'KAHS_Class_Dashboard_Server', 0, winreg.REG_SZ, exe_path) winreg.CloseKey(key) except Exception: pass # ------------------------------------------------------------- # 2. Flask 서버 및 강화된 컴시간 파서 # ------------------------------------------------------------- app = Flask(__name__) CORS(app, resources={r"/*": {"origins": "*"}}) latest_alert = { "id": 0, "title": "", "message": "", "timestamp": "", "sender": "교무실", "active": False } def search_comci_school_code(school_name): """컴시간넷에서 학교 코드를 EUC-KR / UTF-8 방식으로 검색""" headers = { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/122.0.0.0 Safari/537.36', 'Referer': 'http://comci.net:4082/' } # 1. EUC-KR 인코딩 시도 try: encoded = urllib.parse.quote(school_name.encode('euc-kr')) url = f'http://comci.net:4082/38582?98299l{encoded}' res = requests.get(url, headers=headers, timeout=4) match = re.search(r'\((\d{4,7})\)', res.text) if match: return match.group(1) except Exception: pass # 2. UTF-8 인코딩 시도 try: encoded_utf = urllib.parse.quote(school_name.encode('utf-8')) url = f'http://comci.net:4082/38582?98299l{encoded_utf}' res = requests.get(url, headers=headers, timeout=4) match = re.search(r'\((\d{4,7})\)', res.text) if match: return match.group(1) except Exception: pass return None def fetch_comci_timetable(grade=2, class_num=2): """후보 학교명을 순차 검색하여 2학년 2반 시간표 추출""" school_candidates = ['한국항공고', '한국항공고등학교', '태백기계공고', '태백기계공업고'] headers = { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/122.0.0.0 Safari/537.36' } school_code = None for cand in school_candidates: code = search_comci_school_code(cand) if code: school_code = code break if not school_code: return None try: data_url = f'http://comci.net:4082/38582?{school_code}_0_1' res = requests.get(data_url, headers=headers, timeout=5) raw_text = res.text json_match = re.search(r'\{.*\}', raw_text, re.DOTALL) if not json_match: return None # 컴시간 데이터 정제 raw_json_str = json_match.group(0) clean_json_str = re.sub(r'[\x00-\x1f\x7f-\x9f]', '', raw_json_str) data = json.loads(clean_json_str) # 시간표 자료 키값 파싱 (자료481 등) timetable_root = None for k in ['자료481', '자료148', '자료492', '자료']: if k in data: timetable_root = data[k] break if timetable_root and str(grade) in timetable_root: class_data = timetable_root[str(grade)].get(str(class_num)) if class_data: return { "school_code": school_code, "timetable": class_data, "subjects": data.get('자료492', data.get('과목', {})), "teachers": data.get('자료493', data.get('교사', {})) } except Exception: pass return None @app.route('/', methods=['GET']) def index(): return jsonify({ "service": "KAHS 2-2 Dashboard Server", "status": "online", "timestamp": datetime.datetime.now().strftime('%Y-%m-%d %H:%M:%S') }) @app.route('/api/timetable', methods=['GET']) def get_timetable(): parsed = fetch_comci_timetable(grade=2, class_num=2) if parsed: return jsonify({ "success": True, "source": "comci", "data": parsed }) return jsonify({ "success": False, "source": "fallback", "message": "NEIS 연동 전환 필요" }) @app.route('/api/alert', methods=['GET', latest_alert["message"] = req_data.get("message", "") latest_alert["sender"] = req_data.get("sender", "담임교사") latest_alert["timestamp"] = datetime.datetime.now().strftime('%Y-%m-%d %H:%M:%S') latest_alert["active"] = True return jsonify({"success": True, "message": "알림 전송 완료", "alert": latest_alert}) return jsonify(latest_alert) @app.route('/api/alert/dismiss', methods=['POST']) def icon.stop() os._exit(0) def setup_tray(): icon_image = create_tray_icon() menu = pystray.Menu( pystray.MenuItem('한국항공고 2-2 대시보드 서버 (가동 중)', None, enabled=False), pystray.MenuItem('서버 종료', on_exit) ) server_thread.start() setup_tray()
